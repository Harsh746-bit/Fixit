Project Overview & Technology Stack

FixIt MongoDB NoSQL Web Application: A full-stack facilities platform with an interactive web browser dashboard at http://localhost:5002, demonstrating real-time connection status badges, client-side validation, MongoDB collection mutations, status transitions, and automated Postman API verification.

Technology Stack: Node.js, Express.js, MongoDB Atlas (fixit_campus_db ➔ maintenance_issues), Mongoose 8.3, Bootstrap 5 UI, AJAX Fetch Engine, Postman.

Theoretical Background & Concepts

1. NoSQL BSON Document Model

NoSQL document stores model records as flexible, semi-structured BSON documents rather than rigid tabular relations. MongoDB collections store self-contained documents with nested arrays and sub-documents, enabling horizontal scaling without costly multi-table JOIN operations.

2. Mongoose ODM Validation & Cast Error Handling

Mongoose enforces schema rules before data reaches MongoDB:

  • match: Enforces RegEx for email and 10-digit Indian phone numbers.
  • enum: Restricts categories and priorities to pre-approved lists.
  • CastError: Triggered when invalid 24-character hexadecimal ObjectId strings are queried, caught by centralized error middleware to return clean HTTP 400 responses instead of crashing the server.

3. Browser-to-Database End-to-End Flow

The client browser dispatches asynchronous AJAX fetch() requests to Express routes, which pass through validation middleware to Mongoose ODM models. Mongoose executes queries against MongoDB Atlas, returning JSON responses that update the DOM dynamically without page reloads.

const { Issue } = require('../models/Issue');

// @desc    Retrieve all issues from MongoDB collection
// @route   GET /api/issues
exports.getIssues = async (req, res, next) => {
    try {
        const { category, priority, status, search } = req.query;
        const filter = {};

        if (category) filter.category = category;
        if (priority) filter.priority = priority;
        if (status) filter.status = status;

        let issues = await Issue.find(filter);

        if (search && search.trim() !== '') {
            const query = search.trim().toLowerCase();
            issues = issues.filter(item => 
                (item.title && item.title.toLowerCase().includes(query)) ||
                (item.location && item.location.toLowerCase().includes(query)) ||
                (item.description && item.description.toLowerCase().includes(query)) ||
                (item.ticketId && item.ticketId.toLowerCase().includes(query))
            );
        }

        res.status(200).json({
            success: true,
            count: issues.length,
            dbInfo: getDbInfo(),
            data: issues
        });
    } catch (err) {
        next(err);
    }
};

// @desc    Retrieve single issue by MongoDB _id or ticketId
// @route   GET /api/issues/:id
exports.getIssueById = async (req, res, next) => {
    try {
        const { id } = req.params;
        const issue = await Issue.findById(id);

        if (!issue) {
            return res.status(404).json({
                success: false,
                message: `No issue found in MongoDB collection with ID '${id}'.`
            });
        }

        res.status(200).json({
            success: true,
            data: issue
        });
    } catch (err) {
        next(err);
    }
};

/**
 * Validation & Error Handling Middleware
 * Practical 07: Node.js + Express + MongoDB NoSQL Application
 */

const VALID_CATEGORIES = ['Electrical', 'Plumbing', 'HVAC', 'Furniture', 'Internet', 'Civil', 'General'];
const VALID_PRIORITIES = ['Low', 'Medium', 'High', 'Critical'];
const VALID_STATUSES = ['Pending', 'In Progress', 'Resolved'];
const EMAIL_REGEX = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
const PHONE_REGEX = /^[6-9]\d{9}$/;

const validateIssueInput = (req, res, next) => {
    const isPost = req.method === 'POST';
    const { title, category, location, description, reporterName, reporterEmail, reporterPhone, priority, status } = req.body;
    const errors = [];

    // On POST, required fields must be present. On PUT/PATCH, only validate present fields.
    if (isPost || title !== undefined) {
        if (!title || typeof title !== 'string' || title.trim().length < 3) {
            errors.push("Title is required and must be at least 3 characters.");
        }
    }

    if (isPost || category !== undefined) {
        if (!category || !VALID_CATEGORIES.includes(category)) {
            errors.push(`Category must be one of: ${VALID_CATEGORIES.join(', ')}.`);
        }
    }

    if (isPost || location !== undefined) {
        if (!location || typeof location !== 'string' || location.trim().length === 0) {
            errors.push("Campus location is required.");
        }
    }

    if (isPost || description !== undefined) {
        if (!description || typeof description !== 'string' || description.trim().length < 10) {
            errors.push("Description is required and must be at least 10 characters long.");
        }
    }

    if (isPost || reporterName !== undefined) {
        if (!reporterName || typeof reporterName !== 'string' || reporterName.trim().length === 0) {
            errors.push("Reporter name is required.");
        }
    }

    if (isPost || reporterEmail !== undefined) {
        if (!reporterEmail || !EMAIL_REGEX.test(reporterEmail.trim())) {
            errors.push("Valid reporter email address is required.");
        }
    }

    if (reporterPhone) {
        if (!PHONE_REGEX.test(reporterPhone.toString().trim())) {
            errors.push("Reporter phone must be a valid 10-digit Indian mobile number starting with 6, 7, 8, or 9.");
        }
    }

    if (priority && !VALID_PRIORITIES.includes(priority)) {
        errors.push(`Priority must be one of: ${VALID_PRIORITIES.join(', ')}.`);
    }

    if (status && !VALID_STATUSES.includes(status)) {
        errors.push(`Status must be one of: ${VALID_STATUSES.join(', ')}.`);
    }

    if (errors.length > 0) {
        return res.status(400).json({
            success: false,
            message: "Validation failed on user input.",
            errors: errors
        });
    }

    next();
};

const errorHandler = (err, req, res, next) => {
    console.error('[Practical 07 Error Handler]:', err.stack || err.message);

    // Mongoose CastError (invalid ObjectId)
    if (err.name === 'CastError') {
        return res.status(400).json({
            success: false,
            message: `Invalid MongoDB ObjectId: '${err.value}' is not a valid 24-character hexadecimal string.`
        });
    }

    // Mongoose ValidationError
    if (err.name === 'ValidationError') {
        const errorDetails = Object.values(err.errors).map(e => e.message);
        return res.status(400).json({
            success: false,
            message: "Mongoose Schema Validation failed.",
            errors: errorDetails
        });
    }

    // MongoDB Duplicate Key Error (E11000)
    if (err.code === 11000) {
        const field = Object.keys(err.keyValue)[0];
        return res.status(409).json({
            success: false,
            message: `Duplicate key violation: Field '${field}' with value '${err.keyValue[field]}' already exists in database.`
        });
    }

    res.status(err.status || 500).json({
        success: false,
        message: err.message || "Internal server error occurred in database execution."
    });
};

module.exports = {
    validateIssueInput,
    errorHandler,
    VALID_CATEGORIES,
    VALID_PRIORITIES,
    VALID_STATUSES
};

// 1. Check & display live MongoDB connection details
async function checkDatabaseHealth() {
    try {
        const res = await fetch('/api/db-status');
        const json = await res.json();
        const dbBadge = document.getElementById('dbStatusBadge');
        if (json.success && json.dbInfo) {
            const { isConnected, database, collection, provider } = json.dbInfo;
            if (isConnected) {
                dbBadge.innerHTML = `🟢 MongoDB Connected: <strong>${database}</strong> / Collection: <strong>${collection}</strong>`;
                dbBadge.className = 'badge bg-success p-2 text-wrap';
            } else {
                dbBadge.innerHTML = `🟡 Fallback Mode: <strong>${provider}</strong>`;
                dbBadge.className = 'badge bg-warning text-dark p-2 text-wrap';
            }
        }
    } catch (err) {
        console.error('Failed to query db status:', err);
    }
}

// 2. Fetch and render issues & statistics
async function loadDashboard() {
    const categoryFilter = document.getElementById('filterCategory').value;
    const statusFilter = document.getElementById('filterStatus').value;
    const searchQuery = document.getElementById('searchQuery').value.trim();

    const params = new URLSearchParams();
    if (categoryFilter) params.append('category', categoryFilter);
    if (statusFilter) params.append('status', statusFilter);
    if (searchQuery) params.append('search', searchQuery);

    try {
        const res = await fetch(`${API_BASE}?${params.toString()}`);
        const result = await res.json();

        if (result.success) {
            currentIssues = result.data;
            renderIssuesTable(currentIssues);
            loadStatistics();
        } else {
            showAlert('danger', result.message || 'Error fetching issues.');
        }
    } catch (err) {
        console.error('Error fetching issues:', err);
        showAlert('danger', 'Unable to communicate with the server.');
    }
}

// 3. Fetch summary metrics
async function loadStatistics() {
    try {
        const res = await fetch(`${API_BASE}/stats/summary`);
        const json = await res.json();
        if (json.success) {
            document.getElementById('statTotal').innerText = json.data.total;
            document.getElementById('statPending').innerText = json.data.pending;
            document.getElementById('statInProgress').innerText = json.data.inProgress;
            document.getElementById('statResolved').innerText = json.data.resolved;
        }
    } catch (err) {
        console.error('Error fetching statistics:', err);
    }
}
