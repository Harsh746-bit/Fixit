Project Overview & Technology Stack

FixIt Database RESTful API: A production-grade facilities records service connected to MongoDB / MongoDB Atlas, providing full CRUD endpoints, aggregate metrics (GET /api/records/stats/summary), automated validation, a pre-configured Postman Collection v2.1, and cloud deployment on Render.

Theoretical Background & Concepts

1. RESTful Architectural Constraints

Representational State Transfer (REST) adheres to six architectural constraints:

  • Statelessness: Each client request contains all context necessary for execution; the server retains no session state.
  • Uniform Interface: Standardized HTTP verbs (GET, POST, PUT, PATCH, DELETE) and URI identification.
  • Client-Server Decoupling: Separation of storage/data logic from consumer clients.

2. Database CRUD Operations & Schema Modeling

Mongoose models compile schema definitions into strongly typed documents with automatic casting, custom validation rules, and built-in timestamp tracking (createdAt, updatedAt).

3. Cloud Deployment & Postman Automated Verification

Automated Postman test suites assert status codes (pm.response.to.have.status(200)), schema properties, and dynamic variable propagation across requests. Cloud deployment on Render provides an active public HTTPS endpoint.

        hint: "Visit GET / for a comprehensive list of all supported endpoints."
    });
});

// 6. Global Centralized Error Handling Middleware
app.use((err, req, res, next) => {
    console.error('[Practical 06 Global Error]:', err);

    // Mongoose CastError (invalid ObjectId)
    if (err.name === 'CastError') {
        return res.status(400).json({
            success: false,
            message: `Resource not found with invalid identifier format: '${err.value}'`
        });
    }

    // Mongoose ValidationError
    if (err.name === 'ValidationError') {
        const messages = Object.values(err.errors).map(val => val.message);
        return res.status(400).json({
            success: false,
            message: "Mongoose database validation failure.",
            errors: messages
        });
    }

    res.status(err.status || 500).json({
        success: false,
        message: err.message || "Internal Server Error in database operation."
    });
});

// 7. Initialize Database & Start Server
const startServer = async () => {
    await connectDB();
    const server = app.listen(PORT, () => {
        console.log(`=======================================================`);

        console.log(`Health Check:   http://localhost:${PORT}/`);
        console.log(`=======================================================`);
    });
    return server;
};

if (process.env.NODE_ENV !== 'test') {

}

/**
 * Record Controller
 * Practical 06: RESTful API CRUD Handlers
 */

const { Record } = require('../models/Record');
const { getDbStatus } = require('../config/db');

// @desc    Retrieve all records with optional filtering
// @route   GET /api/records
exports.getAllRecords = async (req, res, next) => {
    try {
        const { category, priority, status, search } = req.query;
        const filter = {};

        if (category) filter.category = category;
        if (priority) filter.priority = priority;
        if (status) filter.status = status;

        let records = await Record.find(filter);

        // Optional full-text search across title, location, description
        if (search && search.trim() !== '') {
            const query = search.trim().toLowerCase();
            records = records.filter(r => 
                (r.title && r.title.toLowerCase().includes(query)) ||
                (r.location && r.location.toLowerCase().includes(query)) ||
                (r.description && r.description.toLowerCase().includes(query)) ||
                (r.ticketId && r.ticketId.toLowerCase().includes(query))
            );
        }

        res.status(200).json({
            success: true,
            count: records.length,
            database: getDbStatus(),
            data: records
        });
    } catch (err) {
        next(err);
    }
};

// @desc    Retrieve a single record by ID or ticketId
// @route   GET /api/records/:id
exports.getRecordById = async (req, res, next) => {
    try {
        const { id } = req.params;
        const record = await Record.findById(id);

        if (!record) {
            return res.status(404).json({
                success: false,
                message: `Record with ID/Ticket '${id}' not found in database.`
            });
        }

        res.status(200).json({
            success: true,
            data: record
        });
    } catch (err) {
        next(err);
    }
};

// @desc    Create a new record in database
// @route   POST /api/records
exports.createRecord = async (req, res, next) => {
    try {
        const newRecord = await Record.create(req.body);

        res.status(201).json({
            success: true,
            message: `Record '${newRecord.ticketId || newRecord._id}' successfully created in database.`,
            data: newRecord
        });
    } catch (err) {
        next(err);
    }
};

// @desc    Update an existing record (full or partial)
// @route   PUT /api/records/:id or PATCH /api/records/:id
exports.updateRecord = async (req, res, next) => {
    try {
        const { id } = req.params;
        const updatedRecord = await Record.findByIdAndUpdate(id, req.body, {
            new: true,
            runValidators: true
        });

        if (!updatedRecord) {
            return res.status(404).json({
                success: false,
                message: `Record with ID '${id}' could not be found to update.`
            });
        }

        res.status(200).json({
            success: true,
            message: `Record '${updatedRecord.ticketId || id}' updated successfully.`,
            data: updatedRecord
        });
    } catch (err) {
        next(err);
    }
};

// @desc    Delete a record from database
// @route   DELETE /api/records/:id
exports.deleteRecord = async (req, res, next) => {
    try {
        const { id } = req.params;
        const deletedRecord = await Record.findByIdAndDelete(id);

        if (!deletedRecord) {
            return res.status(404).json({
                success: false,
                message: `Record with ID '${id}' not found. No record was deleted.`
            });
        }

        res.status(200).json({
            success: true,
            message: `Record '${deletedRecord.ticketId || id}' successfully removed from database.`,
            deletedData: deletedRecord
        });
    } catch (err) {
        next(err);
    }
};

// @desc    Retrieve aggregated database statistics
// @route   GET /api/records/stats/summary
exports.getStatistics = async (req, res, next) => {
    try {
        const all = await Record.find({});
        const total = all.length;
        const pending = all.filter(r => r.status === 'Pending').length;
        const inProgress = all.filter(r => r.status === 'In Progress').length;
        const resolved = all.filter(r => r.status === 'Resolved').length;
        const totalEstimatedCost = all.reduce((sum, r) => sum + (Number(r.estimatedCost) || 0), 0);

        const categoryCounts = all.reduce((acc, curr) => {
            acc[curr.category] = (acc[curr.category] || 0) + 1;
            return acc;
        }, {});

        res.status(200).json({
            success: true,
            data: {
                total,
                pending,
                inProgress,
                resolved,
                totalEstimatedCost,
                categoryBreakdown: categoryCounts
            }
        });
    } catch (err) {
        next(err);
    }
};

// @desc    Seed or reset sample records in database
// @route   POST /api/records/seed
exports.seedRecords = async (req, res, next) => {
    try {
        const seeded = await Record.seedDefaults();
        res.status(200).json({
            success: true,
            message: "Database seeded successfully with default FixIt campus records.",
            count: seeded.length,
            data: seeded
        });
    } catch (err) {
        next(err);
    }
};

/**
 * Record Routes
 * Practical 06: RESTful API Endpoint Definitions
 */

const express = require('express');
const router = express.Router();
const recordController = require('../controllers/recordController');
const { validateRecordCreation, validateRecordUpdate } = require('../middleware/validateRecord');

// Static / aggregate routes first
router.get('/stats/summary', recordController.getStatistics);
router.post('/seed', recordController.seedRecords);

// Collection routes
router.route('/')
    .get(recordController.getAllRecords)
    .post(validateRecordCreation, recordController.createRecord);

// Individual record routes by ID
router.route('/:id')
    .get(recordController.getRecordById)
    .put(validateRecordUpdate, recordController.updateRecord)
    .patch(validateRecordUpdate, recordController.updateRecord)
    .delete(recordController.deleteRecord);

module.exports = router;
