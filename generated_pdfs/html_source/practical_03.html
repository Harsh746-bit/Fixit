Theoretical Background & Concepts

1. React Virtual DOM & Declarative Rendering

React maintains an in-memory Virtual DOM representation of UI trees. When state changes occur, React computes the diff between Virtual DOM snapshots using the reconciliation algorithm, updating only mutated real DOM elements for optimal performance.

2. React Hooks & Controlled Form Architecture

useState allows functional components to maintain internal reactive state. Form elements bind their value directly to state variables with onChange handlers, guaranteeing a single source of truth.

3. Props Drilling & Unidirectional Data Flow

Data flows strictly top-down from parent components (App.jsx) to presentation children (IssueCard.jsx) via Props, while user mutations trigger callback functions passed from the parent, ensuring state integrity.

import React, { useState } from 'react';
import Navbar from './components/Navbar';
import Hero from './components/Hero';
import Statistics from './components/Statistics';
import CategorySection from './components/CategorySection';
import HowItWorks from './components/HowItWorks';
import ReportForm from './components/ReportForm';
import IssueDashboard from './components/IssueDashboard';
import DeleteModal from './components/DeleteModal';
import Footer from './components/Footer';
import { initialIssues } from './data/sampleIssues';

export default function App() {
  const [issues, setIssues] = useState(initialIssues);
  const [nextIdNumber, setNextIdNumber] = useState(1007);
  const [deleteTarget, setDeleteTarget] = useState(null);
  const [toast, setToast] = useState({ show: false, title: '', message: '', type: 'success' });

  const showToast = (title, message, type = 'success') => {
    setToast({ show: true, title, message, type });
    setTimeout(() => {
      setToast(prev => ({ ...prev, show: false }));
    }, 4000);
  };

  const handleAddIssue = (newIssue) => {
    // Functional state update: setIssues([...issues, newIssue])
    setIssues(prevIssues => [newIssue, ...prevIssues]);
    setNextIdNumber(prev => prev + 1);
    showToast('Issue Reported Successfully', `Reference ID: ${newIssue.id}`, 'success');

    // Smooth scroll down to dashboard
    const el = document.getElementById('dashboard-section');
    if (el) el.scrollIntoView({ behavior: 'smooth' });
  };

  const handleStatusChange = (issueId, newStatus) => {
    setIssues(prevIssues =>
      prevIssues.map(issue =>
        issue.id === issueId ? { ...issue, status: newStatus } : issue
      )
    );
    showToast('Status Updated', `Ticket ${issueId} changed to "${newStatus}"`, 'info');
  };

  const handleRequestDelete = (issue) => {
    setDeleteTarget(issue);
  };

  const handleConfirmDelete = (issueId) => {
    setIssues(prevIssues => prevIssues.filter(i => i.id !== issueId));
    setDeleteTarget(null);
    showToast('Issue Deleted', `Ticket ${issueId} removed from state.`, 'danger');
  };

  const handleCancelDelete = () => {
    setDeleteTarget(null);
  };

  return (
    <div className="d-flex flex-column min-vh-100">
      <Navbar />
      <Hero />
      <Statistics issues={issues} />
      <CategorySection />
      <HowItWorks />

      <main className="container py-5">
        <div className="row g-4">
          {/* Left Column: Intake Form */}
          <div className="col-lg-5" id="report-section">
            <ReportForm
              onAddIssue={handleAddIssue}
              nextId={`FIX-${nextIdNumber}`}
            />
          </div>

          {/* Right Column: Search, Filter & Issue Cards */}
          <div className="col-lg-7">
            <IssueDashboard
              issues={issues}
              onStatusChange={handleStatusChange}
              onRequestDelete={handleRequestDelete}
            />
          </div>
        </div>
      </main>

      {/* Delete Confirmation Modal */}
      <DeleteModal
        isOpen={Boolean(deleteTarget)}
        issue={deleteTarget}
        onConfirm={handleConfirmDelete}
        onCancel={handleCancelDelete}
      />

      {/* React Toast Notification */}
      {toast.show && (
        <div className="toast-container position-fixed bottom-0 end-0 p-3" style={{ zIndex: 1080 }}>
          <div className="toast show align-items-center text-white bg-charcoal border-0 shadow">
            <div className="d-flex">
              <div className="toast-body d-flex align-items-center gap-2">
                <i
                  className={`fs-5 bi ${
                    toast.type === 'danger'
                      ? 'bi-exclamation-octagon-fill text-danger'
                      : 'bi-check-circle-fill text-safety-orange'
                  }`}
                ></i>
                <div>
                  <div className="fw-bold">{toast.title}</div>
                  <small className="text-light">{toast.message}</small>
                </div>
              </div>
              <button
                type="button"
                className="btn-close btn-close-white me-2 m-auto"
                onClick={() => setToast(prev => ({ ...prev, show: false }))}
                aria-label="Close"
              ></button>
            </div>
          </div>
        </div>
      )}

      <Footer />
    </div>
  );
}

import React, { useState } from 'react';

export default function ReportForm({ onAddIssue, nextId }) {
  const [formData, setFormData] = useState({
    name: '',
    email: '',
    category: '',
    priority: '',
    location: '',
    contact: '',
    description: ''
  });

  const [errors, setErrors] = useState({});
  const [touched, setTouched] = useState({});

  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  const indianMobileRegex = /^[6-9]\d{9}$/;

  const validate = () => {
    const newErrors = {};

    if (!formData.name.trim()) {
      newErrors.name = 'Please enter your name.';
    }

    if (!formData.email.trim()) {
      newErrors.email = 'Please enter your campus email address.';
    } else if (!emailRegex.test(formData.email.trim())) {
      newErrors.email = 'Please enter a valid email address (e.g. student@campus.edu).';
    }

    if (!formData.category) {
      newErrors.category = 'Please select an issue category.';
    }

    if (!formData.priority) {
      newErrors.priority = 'Please select an issue priority level.';
    }

    if (!formData.location.trim()) {
      newErrors.location = 'Please specify the campus location or room number.';
    }

    if (formData.contact.trim() && !indianMobileRegex.test(formData.contact.trim())) {
      newErrors.contact = 'Please enter a valid 10-digit Indian mobile number (e.g. 9876543210).';
    }

    if (!formData.description.trim()) {
      newErrors.description = 'Please describe the problem.';
    } else if (formData.description.trim().length < 10) {
      newErrors.description = `Description must contain at least 10 characters (currently ${formData.description.trim().length}).`;
    }

    return newErrors;
  };

  const handleChange = (e) => {
    const { name, value } = e.target;
    setFormData(prev => ({ ...prev, [name]: value }));
    // Clear error dynamically as user types
    if (errors[name]) {
      setErrors(prev => ({ ...prev, [name]: '' }));
    }
  };

  const handleBlur = (e) => {
    const { name } = e.target;
    setTouched(prev => ({ ...prev, [name]: true }));
    const validationErrors = validate();
    if (validationErrors[name]) {
      setErrors(prev => ({ ...prev, [name]: validationErrors[name] }));
    }
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    const validationErrors = validate();
    setErrors(validationErrors);
    setTouched({
      name: true,
      email: true,
      category: true,
      priority: true,
      location: true,
      contact: true,
      description: true
    });

    if (Object.keys(validationErrors).length > 0) {
      return;
    }

    const todayDate = new Date().toISOString().split('T')[0];
    const newIssue = {
      id: nextId,
      name: formData.name.trim(),
      email: formData.email.trim(),
      category: formData.category,
      priority: formData.priority,
      location: formData.location.trim(),
      contact: formData.contact.trim(),
      description: formData.description.trim(),
      status: 'Pending',
      date: todayDate
    };

    onAddIssue(newIssue);

    // Reset form
    setFormData({
      name: '',
      email: '',
      category: '',
      priority: '',
      location: '',
      contact: '',
      description: ''
    });
    setErrors({});
    setTouched({});
  };

  const handleReset = () => {
    setFormData({
      name: '',
      email: '',
      category: '',
      priority: '',
      location: '',
      contact: '',
      description: ''
    });
    setErrors({});
    setTouched({});
  };

  return (
    <div className="card border rounded-2 p-4 bg-white shadow-sm sticky-lg-top" style={{ top: '90px', zIndex: 10 }}>
      <div className="d-flex align-items-center justify-content-between pb-3 border-bottom mb-3">
        <h2 className="h5 fw-bold text-charcoal mb-0 d-flex align-items-center gap-2">
          <i className="bi bi-file-earmark-plus-fill text-safety-orange"></i>
          Report Campus Issue
        </h2>
        <span className="badge bg-charcoal text-white font-monospace">Next ID: {nextId}</span>
      </div>

      <form onSubmit={handleSubmit} noValidate>
        {/* Student Name */}
        <div className="mb-3">
          <label htmlFor="name" className="form-label fw-semibold text-charcoal small">
            Student / Staff Name <span className="text-danger">*</span>
          </label>
          <input
            type="text"
            className={`form-control ${touched.name && errors.name ? 'is-invalid' : ''}`}
            id="name"
            name="name"
            placeholder="e.g. Rahul Sharma"
            value={formData.name}
            onChange={handleChange}
            onBlur={handleBlur}
          />
          {errors.name && <div className="invalid-feedback small d-block">{errors.name}</div>}
        </div>

        {/* Email */}
        <div className="mb-3">
          <label htmlFor="email" className="form-label fw-semibold text-charcoal small">
            Campus Email <span className="text-danger">*</span>
          </label>
          <input
            type="email"
            className={`form-control ${touched.email && errors.email ? 'is-invalid' : ''}`}
            id="email"
            name="email"
            placeholder="student@example.com"
            value={formData.email}
            onChange={handleChange}
            onBlur={handleBlur}
          />
          {errors.email && <div className="invalid-feedback small d-block">{errors.email}</div>}
        </div>

        {/* Category & Priority */}
        <div className="row g-2 mb-3">
          <div className="col-sm-6">
            <label htmlFor="category" className="form-label fw-semibold text-charcoal small">
              Category <span className="text-danger">*</span>
            </label>
            <select
              className={`form-select ${touched.category && errors.category ? 'is-invalid' : ''}`}
              id="category"
              name="category"
              value={formData.category}
              onChange={handleChange}
              onBlur={handleBlur}
            >
              <option value="" disabled>Select...</option>
              <option value="Electrical">Electrical</option>
              <option value="Plumbing">Plumbing</option>
              <option value="Internet">Internet &amp; Wi-Fi</option>
              <option value="Furniture">Furniture</option>
              <option value="Cleaning">Cleaning</option>
              <option value="Equipment">Equipment</option>
            </select>
            {errors.category && <div className="invalid-feedback small d-block">{errors.category}</div>}
          </div>

          <div className="col-sm-6">
            <label htmlFor="priority" className="form-label fw-semibold text-charcoal small">
              Priority <span className="text-danger">*</span>
            </label>
            <select
              className={`form-select ${touched.priority && errors.priority ? 'is-invalid' : ''}`}
              id="priority"
              name="priority"
              value={formData.priority}
              onChange={handleChange}
              onBlur={handleBlur}

import React from 'react';

export default function IssueCard({ issue, onStatusChange, onRequestDelete }) {
  const getPriorityBadgeClass = (priority) => {
    switch (priority) {
      case 'Critical': return 'badge-priority-Critical';
      case 'High': return 'badge-priority-High';
      case 'Medium': return 'badge-priority-Medium';
      case 'Low': return 'badge-priority-Low';
      default: return 'bg-secondary text-white';
    }
  };

  const getStatusBadgeClass = (status) => {
    switch (status) {
      case 'Pending': return 'badge-status-Pending';
      case 'In Progress': return 'badge-status-InProgress';
      case 'Resolved': return 'badge-status-Resolved';
      default: return 'bg-light text-slate';
    }
  };

  return (
    <article className="issue-card" data-id={issue.id}>
      <div className="d-flex flex-wrap justify-content-between align-items-center gap-2 mb-2">
        <div className="d-flex align-items-center gap-2">
          <span className="badge bg-charcoal font-monospace">{issue.id}</span>
          <span className={`badge ${getPriorityBadgeClass(issue.priority)}`}>{issue.priority} Priority</span>
          <span className="badge bg-light text-slate border">{issue.category}</span>
        </div>
        <div className="d-flex align-items-center gap-2">
          <span className={`badge ${getStatusBadgeClass(issue.status)} font-semibold`}>{issue.status}</span>
        </div>
      </div>

      <h4 className="h6 fw-bold text-charcoal mb-1">{issue.location}</h4>
      <p className="small text-secondary mb-3">{issue.description}</p>

      <div className="d-flex flex-wrap justify-content-between align-items-center gap-2 pt-2 border-top border-light-gray small">
        <div className="text-slate">
          <i className="bi bi-person-circle me-1"></i>{issue.name} &bull; 
          <span className="text-secondary ms-1">{issue.date}</span>
          {issue.contact && (
            <span className="ms-1">&bull; <i className="bi bi-telephone me-1"></i>+91 {issue.contact}</span>
          )}
        </div>

        <div className="d-flex align-items-center gap-2">
          {/* Status update select */}
          <select
            className="form-select form-select-sm"
            style={{ fontSize: '0.82rem', fontWeight: '600', padding: '0.25rem 0.5rem', width: 'auto' }}
            value={issue.status}
            onChange={(e) => onStatusChange(issue.id, e.target.value)}
            aria-label={`Change status for ticket ${issue.id}`}
          >
            <option value="Pending">Pending</option>
            <option value="In Progress">In Progress</option>
            <option value="Resolved">Resolved</option>
          </select>

          {/* Delete button */}
          <button
            type="button"
            className="btn btn-outline-danger btn-sm px-2 py-1"
            onClick={() => onRequestDelete(issue)}
            title="Delete ticket"
            aria-label={`Delete ticket ${issue.id}`}
          >
            <i className="bi bi-trash3"></i>
          </button>
        </div>
      </div>
    </article>
  );
}
