Theoretical Background & Concepts

1. Express.js Middleware Validation Pipeline

In Express, server-side validation is cleanly isolated into middleware functions: Request ➔ validateIssueCreation middleware ➔ issueController ➔ Response. This keeps route handlers focused purely on business logic.

2. Routing & Route Parameters

Express Router (express.Router()) enables modular route definitions. Route parameters (req.params.id) extract target resource identifiers cleanly, while req.query handles filtering without regex URL parsing.

3. Centralized Error Handling Middleware

Express four-argument error middleware (err, req, res, next) catches all unhandled server exceptions, ensuring that even unexpected runtime failures produce uniform, readable JSON responses rather than leaking stack traces.

const EMAIL_REGEX = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
const INDIAN_MOBILE_REGEX = /^[6-9]\d{9}$/;
const ALLOWED_CATEGORIES = ['Electrical', 'Plumbing', 'Internet', 'Furniture', 'Cleaning', 'Equipment'];
const ALLOWED_PRIORITIES = ['Low', 'Medium', 'High', 'Critical'];
const ALLOWED_STATUSES = ['Pending', 'In Progress', 'Resolved'];

/**
 * Middleware: Validates new issue creation body (POST /api/issues)
 */
function validateIssueCreation(req, res, next) {
    const { name, email, category, priority, location, description, contact } = req.body;
    const errors = [];

    // 1. Student / Staff Name
    if (!name || typeof name !== 'string' || !name.trim()) {
        errors.push("Field 'name' is required and cannot be empty.");
    }

    // 2. Email Address
    if (!email || typeof email !== 'string' || !email.trim()) {
        errors.push("Field 'email' is required.");
    } else if (!EMAIL_REGEX.test(email.trim())) {
        errors.push("Field 'email' must be a valid email format.");
    }

    // 3. Category
    if (!category || !ALLOWED_CATEGORIES.includes(category)) {
        errors.push(`Field 'category' is required. Allowed values: ${ALLOWED_CATEGORIES.join(', ')}.`);
    }

    // 4. Priority
    if (!priority || !ALLOWED_PRIORITIES.includes(priority)) {
        errors.push(`Field 'priority' is required. Allowed values: ${ALLOWED_PRIORITIES.join(', ')}.`);
    }

    // 5. Location
    if (!location || typeof location !== 'string' || !location.trim()) {
        errors.push("Field 'location' is required.");
    }

    // 6. Description
    if (!description || typeof description !== 'string' || description.trim().length < 10) {
        errors.push("Field 'description' is required and must contain at least 10 characters.");
    }

    // 7. Optional Contact Number
    if (contact && typeof contact === 'string' && contact.trim()) {
        if (!INDIAN_MOBILE_REGEX.test(contact.trim())) {
            errors.push("Field 'contact' if provided must be a valid 10-digit Indian mobile number (e.g. 9876543210).");
        }
    }

    // If validation fails, return 400 Bad Request
    if (errors.length > 0) {
        return res.status(400).json({
            success: false,
            message: "Validation failed for new issue creation.",
            errors: errors
        });
    }

    // Sanitize and forward
    req.body.name = name.trim();
    req.body.email = email.trim();
    req.body.location = location.trim();
    req.body.description = description.trim();
    if (contact) req.body.contact = contact.trim();

    next();
}

/**
 * Middleware: Validates status update body (PATCH /api/issues/:id/status)
 */
function validateStatusUpdate(req, res, next) {
    const { status } = req.body;

    if (!status || !ALLOWED_STATUSES.includes(status)) {
        return res.status(400).json({
            success: false,
            message: `Invalid status '${status}'. Allowed statuses: ${ALLOWED_STATUSES.join(', ')}.`
        });
    }

    next();
}

module.exports = {
    validateIssueCreation,
    validateStatusUpdate,
    ALLOWED_CATEGORIES,
    ALLOWED_PRIORITIES,
    ALLOWED_STATUSES
};

const fs = require('fs');
const path = require('path');

const DATA_FILE = path.join(__dirname, '..', 'data', 'issues.json');

/**
 * Utility: Read issues from disk
 */
function readIssues() {
    try {
        const raw = fs.readFileSync(DATA_FILE, 'utf8');
        return JSON.parse(raw);
    } catch (err) {
        console.error('[Issue Controller] Error reading JSON database:', err);
        return [];
    }
}

/**
 * Utility: Write issues to disk
 */
function writeIssues(issues) {
    fs.writeFileSync(DATA_FILE, JSON.stringify(issues, null, 2), 'utf8');
}

/**
 * @desc    Get all maintenance issues (supports search, category, status, priority filters)
 * @route   GET /api/issues
 */
function getAllIssues(req, res) {
    try {
        const issues = readIssues();
        const { search, category, status, priority } = req.query;

        let filtered = issues;

        if (search) {
            const s = search.toLowerCase();
            filtered = filtered.filter(i =>
                i.id.toLowerCase().includes(s) ||
                i.location.toLowerCase().includes(s) ||
                i.description.toLowerCase().includes(s) ||
                i.name.toLowerCase().includes(s)
            );
        }

        if (category && category !== 'All') {
            filtered = filtered.filter(i => i.category.toLowerCase() === category.toLowerCase());
        }

        if (status && status !== 'All') {
            filtered = filtered.filter(i => i.status.toLowerCase() === status.toLowerCase());
        }

        if (priority && priority !== 'All') {
            filtered = filtered.filter(i => i.priority.toLowerCase() === priority.toLowerCase());
        }

        return res.status(200).json({
            success: true,
            total: issues.length,
            count: filtered.length,
            data: filtered
        });
    } catch (error) {
        return res.status(500).json({
            success: false,
            message: "Failed to retrieve issues.",
            error: error.message
        });
    }
}

/**
 * @desc    Get single issue by ID
 * @route   GET /api/issues/:id
 */
function getIssueById(req, res) {
    try {
        const { id } = req.params;
        const issues = readIssues();
        const found = issues.find(i => i.id.toUpperCase() === id.toUpperCase());

        if (!found) {
            return res.status(404).json({
                success: false,
                message: `Ticket with ID '${id}' was not found.`
            });
        }

        return res.status(200).json({
            success: true,
            data: found
        });
    } catch (error) {
        return res.status(500).json({
            success: false,
            message: "Error fetching ticket details.",
            error: error.message
        });
    }
}

/**
 * @desc    Create a new maintenance issue
 * @route   POST /api/issues
 */
function createIssue(req, res) {
    try {
        const { name, email, category, priority, location, description, contact } = req.body;
        const issues = readIssues();

        // Calculate next ID
        let maxId = 1000;
        issues.forEach(i => {
            const match = i.id && i.id.match(/^FIX-(\d+)$/);
            if (match) {
                const num = parseInt(match[1], 10);
                if (num > maxId) maxId = num;
            }
        });
        const generatedId = `FIX-${maxId + 1}`;
        const todayDate = new Date().toISOString().split('T')[0];

        const newIssue = {
            id: generatedId,
            name,
            email,
            category,
            priority,
            location,
            description,
            contact: contact || "",
            status: "Pending",
            date: todayDate
        };

        issues.unshift(newIssue);
        writeIssues(issues);

        return res.status(201).json({
            success: true,
            message: "Issue reported successfully.",
            referenceId: generatedId,
            data: newIssue
        });
    } catch (error) {
        return res.status(500).json({
            success: false,
            message: "Error saving new issue.",
            error: error.message
        });
    }
}

/**
 * @desc    Update issue status (Pending | In Progress | Resolved)
 * @route   PATCH /api/issues/:id/status
 */
function updateIssueStatus(req, res) {
    try {
        const { id } = req.params;
        const { status } = req.body;

        const issues = readIssues();
        const targetIndex = issues.findIndex(i => i.id.toUpperCase() === id.toUpperCase());

        if (targetIndex === -1) {
            return res.status(404).json({
                success: false,
                message: `Ticket with ID '${id}' was not found.`
            });
        }

        const oldStatus = issues[targetIndex].status;
        issues[targetIndex].status = status;
        writeIssues(issues);

        return res.status(200).json({
            success: true,
            message: `Ticket '${id}' status updated from '${oldStatus}' to '${status}'.`,
            data: issues[targetIndex]
        });
    } catch (error) {
        return res.status(500).json({
            success: false,
            message: "Error updating ticket status.",
            error: error.message
        });
    }
}

/**
 * ==============================================================================
 * FixIt — Smart Campus Maintenance Management System
 * PRACTICAL 05: Express REST Routes

const express = require('express');
const router = express.Router();
const issueController = require('../controllers/issueController');
const { validateIssueCreation, validateStatusUpdate } = require('../middleware/validation');

// 1. GET /api/issues - Return all issues (with optional filter query parameters)
router.get('/issues', issueController.getAllIssues);

// 2. GET /api/statistics - Return dynamically calculated summary metrics
router.get('/statistics', issueController.getStatistics);

// 3. GET /api/issues/:id - Return a specific issue by ID
router.get('/issues/:id', issueController.getIssueById);

// 4. POST /api/issues - Create a new issue (with validation middleware)
router.post('/issues', validateIssueCreation, issueController.createIssue);

// 5. PATCH /api/issues/:id/status - Update an issue's status
router.patch('/issues/:id/status', validateStatusUpdate, issueController.updateIssueStatus);

// 6. DELETE /api/issues/:id - Delete an issue by ID
router.delete('/issues/:id', issueController.deleteIssue);

module.exports = router;
